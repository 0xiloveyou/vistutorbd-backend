================================================================================
 MATH LEARNING PLATFORM - FULL ENTITY / FIELD / TYPE / CONSTRAINTS REFERENCE
 (MongoDB + Prisma 6.19) - every entity, every field, columnar format
================================================================================

LEGEND
  PK   = primary key
  UQ   = unique
  IX   = indexed
  EN   = enum-like (string constrained by convention)
  EM   = embedded composite
  DEF  = default value
  MDB  = no native varchar; sizes below are practical/conventional, not enforced

================================================================================
 RELATIONSHIP MAP (all string-id based, no Prisma @relation)
================================================================================

User
 +--(1:N) QuizConfig    (userId)    --> Chapter
 +--(1:N) QuizAttempt   (userId)    --> Chapter
 +--(1:N) Bookmark      (userId)    --> Topic
 +--(1:N) Progress      (userId)    --> Topic
 +--(1:N) ChatSession --(1:N) ChatMessage
 +--(1:N) ForumPost   --(1:N) ForumComment
 +--(1:N) Visit  (optional)
 +--(1:N) ModerationReport (reportedBy / reviewedBy)

ClassBand (id) --(1:N)--> ClassLevel (bandId)
ClassLevel (id) --(1:N)--> Category (classLevelId)
                --(1:N)--> Chapter  (classLevelId, +classBandId)
Category (id)   --(1:N self)--> Category (parentId)
                --(1:N)--> Chapter (categoryId)
                --(1:N)--> Topic   (categoryId)
Chapter (id)    --(1:N)--> Topic (chapterId)
                --(1:N)--> Question (chapterId)
                --(1:N)--> QuizConfig / QuizAttempt (chapterId)
                --(1:1)--> LibraryChapter (chapterId)
                             +--(1:1)--> LibraryVisualization (chapterId)

Topic (id)      --(1:N)--> LessonBlock[]  (EMBEDDED)
                --(N:M)--> Visualization  (visualizationIds[])
                --(1:N)--> Question        (topicId)
                --(1:N)--> KnowledgeChunk  (topicId)
                --(1:N)--> Bookmark / Progress (topicId)
                --(N:M)--> QuestionTemplate (topicId)

Visualization (vizId) --(1:N)--> KnowledgeChunk (vizId)
                      <--(N:M)-- Topic.visualizationIds

QuestionTemplate (id) --(1:N)--> Question (templateId)

LearnCarousel   -- standalone, scoped by classLevelIds[] / classBandIds[]
Companion       -- standalone lookup (LessonBlock.companion -> Companion.key)
ChatSession     --(1:N)--> ChatMessage (sessionId)
ForumPost       --(1:N)--> ForumComment (postId)
ModerationReport -- polymorphic soft ref -> chat_message | topic | visualization | question
Visit           --(N:1 optional)--> User (userId)
KnowledgeChunk  -- soft refs -> topicId, chapterId, categoryId, classBandId, vizId, stepIndex
